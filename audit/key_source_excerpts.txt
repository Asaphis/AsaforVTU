    41	const creditWallet = async (userId, amount, walletType = 'main', description = null, reference = null) => {
    42	  const client = await pool.connect();
    43	  
    44	  try {
    45	    await client.query('BEGIN');
    46	
    47	    // Ensure wallet exists
    48	    let wallet = await getWalletByUserId(userId);
    49	    if (!wallet) {
    50	      wallet = await createWallet(userId);
    51	    }
    52	
    53	    // Determine which balance to update
    54	    const balanceField = `${walletType}_balance`;
    55	    const currentBalance = wallet[balanceField] || 0;
    56	    const newBalance = parseFloat(currentBalance) + parseFloat(amount);
    57	
    58	    // Update wallet balance
    59	    await client.query(
    60	      `UPDATE wallets 
    61	       SET ${balanceField} = $1, 
    62	           total_earned = total_earned + $2,
    63	           updated_at = CURRENT_TIMESTAMP 
    64	       WHERE user_id = $3`,
    65	      [newBalance, amount, userId]
    66	    );
    67	
    68	    // Record transaction
    69	    await client.query(
    70	      `INSERT INTO wallet_transactions (user_id, wallet_id, type, amount, balance_after, description, reference)
    71	       VALUES ($1, $2, $3, $4, $5, $6, $7)`,
    72	      [userId, wallet.id, 'credit', amount, newBalance, description, reference]
    73	    );
    74	
    75	    await client.query('COMMIT');
    76	
    77	    // Return updated wallet
    78	    const updatedWallet = await getWalletByUserId(userId);
    79	    return updatedWallet;
    80	  } catch (error) {
    81	    await client.query('ROLLBACK');
    82	    console.error('[Wallet Service] Error crediting wallet:', error);
    83	    throw error;
    84	  } finally {
    85	    client.release();
    86	  }
